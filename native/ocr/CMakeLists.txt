cmake_minimum_required(VERSION 3.20)
project(vse_native_ocr LANGUAGES CXX)
if(APPLE)
 enable_language(OBJCXX)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4.10 REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
set(ORT_INCLUDE_DIR "" CACHE PATH "Pinned official ORT SDK headers")
set(OCR_DEPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/deps" CACHE PATH "Verified downloaded Clipper/JSON source directory")
set(ORT_LIBRARY "" CACHE FILEPATH "Official ORT shared library or Windows import library")
if(NOT EXISTS "${ORT_LIBRARY}")
 message(FATAL_ERROR "ORT_LIBRARY must point to verified official ORT 1.31 runtime/import library")
endif()
add_executable(native-ocr worker.cpp "${OCR_DEPS_DIR}/clipper.cpp")
option(VSE_OCR_BUILD_PROVIDER_TESTS "Build standalone provider-evidence unit tests (never packaged)" OFF)
if(VSE_OCR_BUILD_PROVIDER_TESTS)
 add_executable(native-provider-evidence-test tests/provider_evidence_test.cpp)
 target_include_directories(native-provider-evidence-test PRIVATE "${OCR_DEPS_DIR}")
 add_executable(native-protocol-output-test tests/protocol_output_test.cpp)
 target_compile_definitions(native-protocol-output-test PRIVATE VSE_NATIVE_PROTOCOL_TEST=1)
 target_link_libraries(native-protocol-output-test PRIVATE Threads::Threads)
endif()
option(VSE_OCR_BUILD_PRECISION_DIAGNOSTIC "Build separate, never-packaged precision evidence target" OFF)
set(OCR_TARGETS native-ocr)
if(VSE_OCR_BUILD_PRECISION_DIAGNOSTIC)
 add_executable(native-ocr-precision EXCLUDE_FROM_ALL worker.cpp "${OCR_DEPS_DIR}/clipper.cpp")
 target_compile_definitions(native-ocr-precision PRIVATE VSE_OCR_PRECISION_DIAGNOSTIC=1)
 list(APPEND OCR_TARGETS native-ocr-precision)
endif()
foreach(OCR_TARGET IN LISTS OCR_TARGETS)
if(APPLE)
 target_sources(${OCR_TARGET} PRIVATE metal_capture.mm)
 set_target_properties(${OCR_TARGET} PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED ON)
 set_source_files_properties(metal_capture.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
 target_link_libraries(${OCR_TARGET} PRIVATE "-framework Foundation" "-framework Metal")
else()
 target_sources(${OCR_TARGET} PRIVATE metal_capture_stub.cpp)
endif()
target_include_directories(${OCR_TARGET} PRIVATE "${ORT_INCLUDE_DIR}" "${OCR_DEPS_DIR}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(${OCR_TARGET} PRIVATE ${OpenCV_LIBS} "${ORT_LIBRARY}" Threads::Threads)
if(MSVC)
 target_compile_options(${OCR_TARGET} PRIVATE /O2 /fp:strict /utf-8)
else()
 target_compile_options(${OCR_TARGET} PRIVATE -O2 -ffp-contract=off -fno-fast-math)
 if(UNIX AND NOT APPLE)
  target_link_options(${OCR_TARGET} PRIVATE -Wl,--gc-sections)
 endif()
 if(APPLE)
  set_target_properties(${OCR_TARGET} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
 else()
  set_target_properties(${OCR_TARGET} PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
 endif()
endif()
endforeach()
