cmake_minimum_required(VERSION 3.20)
project(vse_native_ocr LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4.10 REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
set(ORT_INCLUDE_DIR "" CACHE PATH "Pinned official ORT SDK headers")
set(OCR_DEPS_DIR "${CMAKE_CURRENT_SOURCE_DIR}/deps" CACHE PATH "Verified downloaded Clipper/JSON source directory")
set(ORT_LIBRARY "" CACHE FILEPATH "Official ORT shared library or Windows import library")
if(NOT EXISTS "${ORT_LIBRARY}")
 message(FATAL_ERROR "ORT_LIBRARY must point to verified official ORT 1.31 CPU runtime/import library")
endif()
add_executable(native-ocr worker.cpp "${OCR_DEPS_DIR}/clipper.cpp")
target_include_directories(native-ocr PRIVATE "${ORT_INCLUDE_DIR}" "${OCR_DEPS_DIR}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(native-ocr PRIVATE ${OpenCV_LIBS} "${ORT_LIBRARY}" Threads::Threads)
if(MSVC)
 target_compile_options(native-ocr PRIVATE /O2 /fp:strict /utf-8)
else()
 target_compile_options(native-ocr PRIVATE -O2 -ffp-contract=off -fno-fast-math)
 if(UNIX AND NOT APPLE)
  target_link_options(native-ocr PRIVATE -Wl,--gc-sections)
 endif()
 if(APPLE)
  set_target_properties(native-ocr PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE BUILD_RPATH "@loader_path" INSTALL_RPATH "@loader_path")
 else()
  set_target_properties(native-ocr PROPERTIES BUILD_WITH_INSTALL_RPATH TRUE BUILD_RPATH "$ORIGIN" INSTALL_RPATH "$ORIGIN")
 endif()
endif()
