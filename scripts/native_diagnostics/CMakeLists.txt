cmake_minimum_required(VERSION 3.20)
project(vse_native_diagnostics LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 4.10 EXACT REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
set(ORT_INCLUDE_DIR "" CACHE PATH "Verified, existing ORT 1.31.0 headers")
set(ORT_LIBRARY "" CACHE FILEPATH "Verified, existing ORT 1.31.0 shared library")
set(OCR_DEPS_DIR "" CACHE PATH "Verified existing Clipper and JSON sources")
if(WIN32)
  message(FATAL_ERROR "This isolated diagnostic supports Linux validation and macOS execution only")
endif()
foreach(required "${ORT_INCLUDE_DIR}/onnxruntime_c_api.h" "${ORT_LIBRARY}" "${OCR_DEPS_DIR}/clipper.cpp" "${OCR_DEPS_DIR}/json.hpp")
  if(NOT EXISTS "${required}")
    message(FATAL_ERROR "Missing offline input: ${required}; no download/build fallback")
  endif()
endforeach()
add_executable(vse-provider-diagnostic runner.cpp "${OCR_DEPS_DIR}/clipper.cpp")
target_include_directories(vse-provider-diagnostic PRIVATE "${ORT_INCLUDE_DIR}" "${OCR_DEPS_DIR}" ${OpenCV_INCLUDE_DIRS})
target_link_libraries(vse-provider-diagnostic PRIVATE ${OpenCV_LIBS} "${ORT_LIBRARY}" Threads::Threads ${CMAKE_DL_LIBS})
target_compile_options(vse-provider-diagnostic PRIVATE -O2 -ffp-contract=off -fno-fast-math)
